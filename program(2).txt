function submitVote(optionId) {
// Simulate AJAX POST. In production:
// $.post('/api/vote', { poll: poll.id, option: optionId }, ...)
setTimeout(function () {
// "Server" updates the vote count
poll.options.forEach(function (o) {
if (o.id === optionId) o.votes++;
});
localStorage.setItem(poll.id, optionId);
// Smooth UI transition
$('#optionsBox').slideUp(400, function () {
$(this).empty();
$('#voted').slideDown(300);
showResults();
});
}, 400);
}
function showResults() {
var total = poll.options.reduce(function (s, o) { return s + o.votes; }, 0);
var html = '';
poll.options.forEach(function (o) {
var pct = total === 0 ? 0 : Math.round((o.votes / total) * 100);
html += '<div class="result-bar">' +
'<div class="fill" data-pct="' + pct + '"></div>' +
'<span class="label">' + o.text + ' (' + o.votes + ')</span>' +
'<span class="pct">' + pct + '%</span>' +
'</div>';
});
$('#results').html(html).slideDown(400);
// Animate the bars
setTimeout(function () {
$('.fill').each(function () { $(this).css('width', $(this).data('pct') + '%'); });
}, 100);
$('#reset').show();
}
function resetVote() {
localStorage.removeItem(poll.id);
location.reload();
}
